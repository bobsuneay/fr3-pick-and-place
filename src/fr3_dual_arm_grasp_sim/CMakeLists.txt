cmake_minimum_required(VERSION 3.16)
project(fr3_dual_arm_grasp_sim)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
add_library(fr3_dual_arm_assisted_grasp SHARED src/assisted_grasp.cpp)
target_compile_features(fr3_dual_arm_assisted_grasp PUBLIC cxx_std_17)
target_include_directories(fr3_dual_arm_assisted_grasp PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(fr3_dual_arm_assisted_grasp PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(fr3_dual_arm_assisted_grasp ${GAZEBO_LIBRARIES})
ament_target_dependencies(fr3_dual_arm_assisted_grasp gazebo_ros rclcpp std_srvs)
install(TARGETS fr3_dual_arm_assisted_grasp LIBRARY DESTINATION lib)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
